SMART CITY TRAFFIC & TRANSIT ANALYTICS — MANUAL DEMO COMMANDS
================================================================
These are the SAME steps that scripts/run_pipeline.sh runs automatically,
broken out one command at a time so you can paste them into a terminal
yourself, watch each one complete, and screenshot the output for your
report/viva. Nothing in this file executes on its own.

Shell: any Unix-style shell — a normal terminal on macOS/Linux, or
Git Bash on Windows (NOT PowerShell/cmd). WSL2 also works.
Working directory: cd into wherever you cloned this project, e.g.
/c/Projects/BDA_PROJECT on Windows Git Bash, or ~/BDA_PROJECT on macOS/Linux.

------------------------------------------------------------------
0. ONE-TIME SETUP / START THE CLUSTER
------------------------------------------------------------------
# Make sure Docker Desktop (or the Docker daemon, on Linux) is running first
# - `docker info` should succeed with no error. If it's not running, start
# Docker Desktop from the Start menu / Applications and wait ~30s, then:

python data/generate_data.py
# ~160k GPS pings, ~60k RFID scans, 12 routes, 18 stops, 14 simulated days

docker compose up -d --build
# (this also builds + starts the dashboard container - it'll show setup
# prompts at http://localhost:8501 until sections 6-7 below are done)

# Wait ~30-60s, then confirm all 8 containers are healthy:
docker compose ps

# On Windows/Git Bash, avoid it mangling /smartcity/... paths into
# C:/smartcity/... before they reach docker exec (harmless no-op on
# macOS/Linux, where this doesn't happen):
export MSYS_NO_PATHCONV=1
export MSYS2_ARG_CONV_EXCL="*"

docker exec namenode hdfs dfsadmin -safemode wait

# Make sure a NodeManager has registered before submitting jobs:
docker exec resourcemanager yarn node -list


------------------------------------------------------------------
1. HDFS — ADDING DIRECTORIES AND FILES
------------------------------------------------------------------
docker exec namenode hdfs dfs -mkdir -p /smartcity/raw_gps
docker exec namenode hdfs dfs -mkdir -p /smartcity/raw_rfid

docker cp data/gps_telemetry.csv namenode:/tmp/gps_telemetry.csv
docker cp data/rfid_ticketing.csv namenode:/tmp/rfid_ticketing.csv

docker exec namenode hdfs dfs -put -f /tmp/gps_telemetry.csv /smartcity/raw_gps/
docker exec namenode hdfs dfs -put -f /tmp/rfid_ticketing.csv /smartcity/raw_rfid/


------------------------------------------------------------------
2. HDFS — RETRIEVING / VERIFYING FILES
------------------------------------------------------------------
docker exec namenode hdfs dfs -ls /smartcity/raw_gps/
docker exec namenode hdfs dfs -ls /smartcity/raw_rfid/
docker exec namenode hdfs dfs -cat /smartcity/raw_gps/gps_telemetry.csv | head -n 5


------------------------------------------------------------------
3. HDFS — DELETING A FILE, THEN A DIRECTORY (explicit demo)
------------------------------------------------------------------
# Create a throwaway directory + file so there's something to delete:
docker exec namenode hdfs dfs -mkdir -p /smartcity/tmp_test
docker exec namenode bash -c "echo 'sample line for the delete demo' > /tmp/sample.txt"
docker exec namenode hdfs dfs -put -f /tmp/sample.txt /smartcity/tmp_test/sample.txt

# Show it exists:
docker exec namenode hdfs dfs -ls /smartcity/tmp_test/

# Delete the file, then the (now empty) directory:
docker exec namenode hdfs dfs -rm /smartcity/tmp_test/sample.txt
docker exec namenode hdfs dfs -rmdir /smartcity/tmp_test

# Confirm it's gone (this should print nothing / "No such file or directory"):
docker exec namenode hdfs dfs -ls /smartcity/tmp_test

# Also demonstrates recursive delete of a non-empty directory (used later
# to clear old MapReduce output before re-running a job):
docker exec namenode hdfs dfs -rm -r -f /smartcity/processed_traffic
docker exec namenode hdfs dfs -rm -r -f /smartcity/processed_traffic_hourly
docker exec namenode hdfs dfs -rm -r -f /smartcity/processed_ridership


------------------------------------------------------------------
4. MAPREDUCE — RUN ALL THREE STREAMING JOBS
------------------------------------------------------------------
docker cp mapreduce/traffic/mapper.py namenode:/tmp/traffic_mapper.py
docker cp mapreduce/traffic/reducer.py namenode:/tmp/traffic_reducer.py
docker cp mapreduce/traffic_hourly/mapper.py namenode:/tmp/traffic_hourly_mapper.py
docker cp mapreduce/ridership/mapper.py namenode:/tmp/ridership_mapper.py
docker cp mapreduce/ridership/reducer.py namenode:/tmp/ridership_reducer.py

# Job 1 - Traffic congestion (per-route average speed + congestion level):
docker exec namenode hadoop jar /opt/hadoop-3.2.1/share/hadoop/tools/lib/hadoop-streaming-3.2.1.jar -files /tmp/traffic_mapper.py,/tmp/traffic_reducer.py -mapper "python3 traffic_mapper.py" -reducer "python3 traffic_reducer.py" -input /smartcity/raw_gps -output /smartcity/processed_traffic

# Job 2 - Traffic hourly profile (route x hour x weather x weekday/weekend -
# the ML training set; note it reuses traffic_reducer.py, same file as job 1):
docker exec namenode hadoop jar /opt/hadoop-3.2.1/share/hadoop/tools/lib/hadoop-streaming-3.2.1.jar -files /tmp/traffic_hourly_mapper.py,/tmp/traffic_reducer.py -mapper "python3 traffic_hourly_mapper.py" -reducer "python3 traffic_reducer.py" -input /smartcity/raw_gps -output /smartcity/processed_traffic_hourly

# Job 3 - Ridership (hourly boardings per stop, by passenger type):
docker exec namenode hadoop jar /opt/hadoop-3.2.1/share/hadoop/tools/lib/hadoop-streaming-3.2.1.jar -files /tmp/ridership_mapper.py,/tmp/ridership_reducer.py -mapper "python3 ridership_mapper.py" -reducer "python3 ridership_reducer.py" -input /smartcity/raw_rfid -output /smartcity/processed_ridership

# Verify all 3 jobs' output:
docker exec namenode hdfs dfs -ls /smartcity/processed_traffic/
docker exec namenode hdfs dfs -cat /smartcity/processed_traffic/part-00000 | head -n 10
docker exec namenode hdfs dfs -ls /smartcity/processed_traffic_hourly/
docker exec namenode hdfs dfs -cat /smartcity/processed_traffic_hourly/part-00000 | head -n 10
docker exec namenode hdfs dfs -ls /smartcity/processed_ridership/
docker exec namenode hdfs dfs -cat /smartcity/processed_ridership/part-00000 | head -n 10

# Optional: watch the jobs in the browser while/after they run:
#   ResourceManager UI: http://localhost:8088
#   NameNode UI:         http://localhost:9870


------------------------------------------------------------------
5. RETRIEVE MAPREDUCE RESULTS BACK TO LOCAL DISK
------------------------------------------------------------------
docker exec namenode bash -c "rm -rf /tmp/processed_traffic /tmp/processed_traffic_hourly /tmp/processed_ridership && hdfs dfs -get /smartcity/processed_traffic /tmp/processed_traffic && hdfs dfs -get /smartcity/processed_traffic_hourly /tmp/processed_traffic_hourly && hdfs dfs -get /smartcity/processed_ridership /tmp/processed_ridership"

docker cp namenode:/tmp/processed_traffic output/processed_traffic
docker cp namenode:/tmp/processed_traffic_hourly output/processed_traffic_hourly
docker cp namenode:/tmp/processed_ridership output/processed_ridership


------------------------------------------------------------------
6. MONGODB — LOAD DATA, BUILD INDEXES, RUN ANALYSIS
------------------------------------------------------------------
python -m pip install -r mongo/requirements.txt

python mongo/load_data.py

# Sanity check the document counts (12 / 1728 / 18 / 18 with the default
# data generator settings):
docker exec mongodb mongosh smartcity --quiet --eval "['routes_summary','route_hourly_profile','stop_ridership','transit_stops'].forEach(c => print(c + ': ' + db.getCollection(c).countDocuments()))"

# Run all 6 aggregation pipelines and print their results:
docker exec -i mongodb mongosh smartcity < mongo/aggregations.js

# Optional: open Mongo Express to browse collections/indexes visually:
#   http://localhost:8081
